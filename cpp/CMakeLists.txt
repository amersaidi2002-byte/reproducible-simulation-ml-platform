cmake_minimum_required(VERSION 3.20)
project(simml_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ORT SDK root with include/ and lib/")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_executable(simml_predict predict.cpp)
target_include_directories(simml_predict PRIVATE "${ORT_INCLUDE}")
target_link_libraries(simml_predict PRIVATE "${ORT_LIBRARY}")
target_compile_options(simml_predict PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(simml_predict PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
